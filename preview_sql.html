⚡ Quick Queries
Customers
SELECT * FROM customers All customer records SELECT id, name, phone FROM customers WHERE email IS NOT NULL Customers with email SELECT COUNT(*) AS total_customers FROM customers Customer count
Tables
SELECT * FROM tables All table statuses SELECT * FROM tables WHERE status = 'Available' Available tables only SELECT status, COUNT(*) AS count FROM tables GROUP BY status Table status summary
Menu
SELECT * FROM menu ORDER BY price DESC All menu items, most expensive first SELECT * FROM menu WHERE cat = 'Veg' Veg items only SELECT cat, COUNT(*) AS items, AVG(price) AS avg_price FROM menu GROUP BY cat Category summary with averages SELECT * FROM menu WHERE price > 300 ORDER BY price Premium items (price > ₹300)
Orders
SELECT * FROM orders All orders SELECT * FROM orders WHERE status = 'Active' Active orders only SELECT id, custName, tableId, status FROM orders ORDER BY id DESC Orders summary (latest first) SELECT custName, COUNT(*) AS order_count FROM orders GROUP BY custName ORDER BY order_count DESC Orders per customer
Billing
SELECT * FROM bills All bills SELECT custName, final AS total_paid FROM bills ORDER BY final DESC Highest spenders SELECT SUM(final) AS total_revenue, AVG(final) AS avg_bill, COUNT(*) AS bill_count FROM bills Revenue summary SELECT custName, SUM(final) AS lifetime_value FROM bills GROUP BY custName ORDER BY lifetime_value DESC Customer lifetime value
Reservations
SELECT * FROM reservations All reservations SELECT * FROM reservations WHERE status = 'Confirmed' Confirmed bookings only SELECT custName, COUNT(*) AS bookings FROM reservations GROUP BY custName Bookings per customer
Joins & Analytics
SELECT c.name, c.phone, o.id AS order_id, o.status FROM customers AS c JOIN orders AS o ON c.id = o.custId Customers joined with their orders SELECT o.custName, b.final AS billed, o.tableId FROM orders AS o JOIN bills AS b ON o.id = b.ordId Orders joined with billing data SELECT custName, COUNT(*) AS reservations, MAX(dt) AS latest FROM reservations GROUP BY custName ORDER BY reservations DESC Reservation frequency analysis
